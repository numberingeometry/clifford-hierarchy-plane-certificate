\begin{finitefact}{cert:stationary-cs-ccz-gates-and-complete-affine-flags}{Stationary gates $\mathrm{CS}_{yz}^a\CCZ^b$ and complete affine flags}{Lemma~\ref{lem:tool-bridge}}
The gates $G=\mathrm{CS}_{yz}^a\CCZ^b$ with $a,b\in\F_2$, that is, $I$, $\mathrm{CS}_{yz}$, $\CCZ$ and $\mathrm{CS}_{yz}\CCZ$, are stationary, with $135$, $3$, $15$ and $3$ GSC pairs. Let $\pi$ be the permutation of $CGC^\dagger$, where $C\in\C_2(3)$ maps the Pauli MASA of an input Lagrangian of $G$ to the diagonal matrices. Then for every affine permutation $\alpha$ of $\F_2^3$, if $\alpha\pi$ and the translations preserve a common complete affine flag, so do $\alpha$, $\pi$ and the translations.
\end{finitefact}

\begin{finitefact}{cert:three-qubit-flag-preserving-involution-rays}{Three-qubit involutions with a flag-preserving permutation}{Proposition~\ref{gauss-lem:exceptional}}
Let $M$ be a gate of the form~\eqref{core:eq:row} on three qubits whose permutation preserves the flag $0<\Span\{e_z\}<\Span\{e_y,e_z\}<\F_2^3$, and let $H_*$ be as in the proof of Proposition~\ref{gauss-lem:exceptional}. Then $[M]\in\C_3(3)$, or $[M]$ is conjugate to $[H_*]$ by an affine--diagonal Clifford gate.
\end{finitefact}

\begin{finitefact}{cert:cliffords-commuting-with-hstar}{Cliffords commuting with $H_*$}{Lemma~\ref{gauss-lem:three-correction-compatibility}}
With $H_*$ as above, a Hermitian gate $K\in\C_2(3)$ commutes with $H_*$ if and only if $K=\pm(-1)^{ax+bxy+c(1+x)z}$ with $a,b,c\in\F_2$. The traceless ones are $\pm Z_x$ and $\pm$ two non-Pauli gates whose supports span $L_Z$.
\end{finitefact}

\begin{finitefact}{cert:clifford-involution-groups-in-no-pauli-masa}{Clifford involution groups in no Pauli MASA}{Lemma~\ref{lem:m-M-1}}
Let $\mathcal G\cong\F_2^r$ be a group, modulo signs, of commuting traceless Hermitian three-qubit Clifford involutions and $I$, in which every $g\ne I$ is a Pauli or has four-term support spanning a Lagrangian, and which lies in no Pauli MASA\@. Then, after a left Clifford conjugation and a change of generators, $\mathcal G$ is generated by
\[
\hat X_1,\hat X_2,\qquad\text{or}\qquad \hat X_1,\hat X_2,Z_3,\qquad\text{or}\qquad \hat X_1,\hat X_2,\hat X_3,
\]
where $\hat X_j=QX_jQ$ and $Q=\CCZ$.
\end{finitefact}

\begin{finitefact}{cert:second-difference-equations-for-phases}{Second-difference equations for phases}{Lemma~\ref{lem:m-M-1}}
Let $f:\F_2^3\to\mathbb R/\mathbb Z$, let $T$ be a symmetric binary $3\times3$ matrix, and let $\Theta$ be as in the proof of Lemma~\ref{lem:m-M-1}. Then:
\[
\begin{array}{c|c|c}
T&\text{equations}&\text{consequence}\\\hline
\operatorname{rank}T=3&\Delta_k\Delta_af=0\ (k\cdot Ta=0)&4\Delta_jf=0\ (j=1,2,3)\\
\operatorname{rank}T=3\text{ or }\operatorname{im}T=\Span\{e_1,e_2\},\ T\ne\Theta&\Delta_{\Theta Ta}\Delta_af=0\ (\text{all }a)&4\Delta_jf=0\ (j=1,2)\\
T=\Theta&\Delta_{\Theta Ta}\Delta_af=0\ (\text{all }a)&2\Delta_jf=0\ (j=1,2)
\end{array}
\]
\end{finitefact}

\begin{finitefact}{cert:r4-representatives-up-to-clifford-conjugacy}{Representatives of $\mathscr R_4$ up to Clifford conjugacy}{Lemma~\ref{app:finite-endpoint}}
A list of $1{,}020$ gates of the form~\eqref{core:eq:row} on four qubits contains a Clifford conjugate of every $h\in\mathscr R_4$. Exactly ten listed $M$ have $|N(M)|-1\ge63$, and they lie in $\C_2(4)$.
\end{finitefact}

\begin{finitefact}{cert:plane-meeting-fixed-pauli-labels}{A plane meeting the fixed Pauli labels}{Lemma~\ref{app:finite-endpoint}}
For each of the other $1{,}010$ listed $M$ there is a plane $\Pi_M\le V_4$ with
\[
N(h)\cap\Pi_M\ne\{0\}\qquad\text{for all }h\in\{M\}\cup\mathcal K(M),
\]
where $N(h)=\{s:hP_sh^\dagger=\pm P_s\}$.
\end{finitefact}

\begin{finitefact}{cert:pauli-terms-and-cubic-coefficient}{Pauli terms and the cubic coefficient}{Lemma~\ref{app:return-single}}
Let $D=\pi_0\,i^{axy+bxz+cyz+dxyz}$ with $\pi_0\in\{I,\pi\}$, $a,b,c\in\F_2$ and $d\in\Z/4$. (a) If $d$ is odd, then $D^\dagger PD$ has at least eight Pauli terms for every non-diagonal three-qubit Pauli $P$. (b) If $d$ is even, then for every $p\in\F_2^3$ the labels $v$ with $DT_pP_vT_p^\dagger D^\dagger\in\C_2(3)$ form a subspace of dimension at least five.
\end{finitefact}

\begin{finitefact}{cert:normal-form-of-exceptional-balanced-configurations}{Normal form of the triples $(\rho,u,D)$}{Lemma~\ref{app:return-balanced}}
Let $\tau_0(x,y,z)=(x,y,z+xy)$, and let $\rho$ be a permutation of $\F_2^3$ that is affine or affinely equivalent to $\tau_0$. Let $u\ne0$ and let $D$ be an affine plane with $D+u=D$ such that $\tau\rho$, where $\tau(x)=x+u\,[x\in D]$, is again affine or affinely equivalent to $\tau_0$. If fewer than two independent $c$ with $c\cdot u=0$ make $c\cdot\rho$ affine, then after replacing $(\rho,u,D)$ by $(\alpha\rho\beta,\bar\alpha u,\alpha(D))$ for affine $\alpha,\beta$, with $\bar\alpha$ the linear part of $\alpha$, in coordinates $(s,y,z)$ we have $\rho(s,y,z)=(s,y,z+sy)$, $u=e_y$, and $D=\{s=1\}$ or $D=\{z=1\}$.
\end{finitefact}

\begin{finitefact}{cert:balanced-rank-two-branch}{The case $\operatorname{rank}E=2$}{Lemma~\ref{app:return-balanced}}
In Case~2b of the proof of Lemma~\ref{app:return-balanced}: for $\ell=z$ and the correction $h=WX_sW^\dagger$, the monomial gate $Z_1h$, in the basis $\ket\pm\otimes\ket x$, has a permutation of cycle type $2^26^2$, so this case does not occur. For $\ell=s$, the Pauli-MASA condition on the off-diagonal block makes every $\operatorname{diag}\theta_t$ Clifford, and among the $64$ choices of $a,b,c,d,e,f\in\F_2$ the three unipotence conditions hold exactly when $a=b=c=e=0$ and $d=f$.
\end{finitefact}

\begin{finitefact}{cert:mixed-blocks-with-two-input-lagrangians}{Mixed blocks with two input Lagrangians}{Lemma~\ref{app:return-multiple}}
Let $D=\pi_0\,i^{axy+bxz+cyz+dxyz}$ with $\pi_0\in\{I,\pi\}$, $a,b,c\in\F_2$ and $d\in\Z/4$, a block form of Lemma~\ref{app:return-block}\textup{(4)} with $p'=0$. If $|X(D)|\ge2$, then $D\in\C_3(3)$.
\end{finitefact}

\begin{finitefact}{cert:block-form-clifford-classes}{Clifford classes of the block forms}{Lemma~\ref{app:return-multiple}}
Under $D\mapsto C_1DC_2$ with $C_1,C_2\in\C_2(3)$, the $512$ block forms $D$ of Lemma~\ref{app:return-block}\textup{(4)} fall into $30$ classes. Ten classes are third-level, and every correction $DP_vD^\dagger$ of the other twenty is third-level. We call the listed representative of a class its normal form. The third-level classes containing a form with $p'=0$ are those of the four gates $\mathrm{CS}_{yz}^a\CCZ^b$.
\end{finitefact}

\begin{finitefact}{cert:invariant-sets-of-input-lagrangians}{Invariant sets of input Lagrangians}{Lemma~\ref{app:return-multiple}}
Let $A$ be a non-third-level normal form of \textup{(\ref{cert:block-form-clifford-classes})}, and let $D$ be a Clifford conjugate of one of the four gates $\mathrm{CS}_{yz}^a\CCZ^b$. If $X(D)$ is invariant under $J_A$, then $L_Z\in X(D)$.
\end{finitefact}

\begin{finitefact}{cert:reduction-to-common-input-pair-domain}{Reduction of $B_0$ to $256$ gates}{Lemma~\ref{app:return-multiple}}
Let $A$ be a non-third-level normal form with $T$-factor $T_{p_A}$, and $B_0=\Pi_\rho\,i^{g}T_{p_A}$ with $g:\F_2^3\to\Z/4$ and $\rho$ affine or affinely equivalent to $\pi$ of Lemma~\ref{app:return-block}\textup{(4)}. Let $\rho_0$ be the identity, and for $0\ne k\in\F_2^3$ let $\rho_k=\lambda_k^{-1}\pi\lambda_k$ with $\lambda_k$ linear and $\lambda_k(k)=(0,0,1)$. Modulo left multiplication by monomial Clifford gates, $B_0$ is one of the $256$ gates $\Pi_{\rho_k}\,i^{axy+bxz+cyz+dxyz}\,T_{p_A}$ with $k\in\F_2^3$, $a,b,c\in\F_2$ and $d\in\Z/4$, or $\rho(x+k)+\rho(x)$ does not depend on $x$ exactly for $k\in\{0,(1,1,1)\}$, and then some $B_0P_vA^\dagger\notin\C_3(3)$.
\end{finitefact}

\begin{finitefact}{cert:block-pair-domains}{Pair domains for blocks}{Lemma~\ref{app:return-multiple}}
(a) Let $A$ be a non-third-level normal form of \textup{(\ref{cert:block-form-clifford-classes})} and $B_0$ one of the $256$ gates of \textup{(\ref{cert:reduction-to-common-input-pair-domain})} for this $A$ ($5{,}120$ pairs). For the $360$ pairs with every $D_v=B_0P_vA^\dagger\in\C_3(3)$, each $D_v=G_vY_v$ with $G_v\in\C_2(3)$ and $Y_v$ a Clifford conjugate of some $\mathrm{CS}_{yz}^a\CCZ^b$, and $\mathcal B(A,FG_0^{-1}B_0)$ holds for every $F\in\C_2(3)$ with $\gamma_F\gamma_{G_0^{-1}G_v}e$ unipotent for all $v\in V_3$ and $e\in E_{Y_v}$.

(b) Let $A$ be a third-level normal form and $D$ a Clifford conjugate of some $\mathrm{CS}_{yz}^a\CCZ^b$ with $X(D)$ invariant under $E_A$ ($13{,}960$ pairs, reduced to $2{,}076$). Then $\mathcal B(A,FDA)$ holds for every $F\in\C_2(3)$ with $\gamma_Fef$ unipotent for all $e\in E_D$ and $f\in E_A$.
\end{finitefact}

\begin{finitefact}{cert:s6-two-group-criterion}{A $2$-group criterion in $S_6$}{Theorem~\ref{lem:fib-sharp-small}}
Let $a,b,c\in\Sp(4,2)\cong S_6$. If $\langle a,cb^{-1}\rangle$, $\langle b,ca^{-1}\rangle$ and $\langle c,ba^{-1}\rangle$ are $2$-groups, then so is $\langle a,b,c\rangle$.
\end{finitefact}

\begin{finitefact}{cert:s6-sylow-cosets-and-three-element-subsets}{Sylow cosets in $S_6$ and three-element subsets}{Theorem~\ref{lem:fib-sharp-small}}
Let $F\subseteq\Sp(4,2)\cong S_6$. If every subset of $F$ with at most three elements lies in a right coset $Pg$ of a Sylow $2$-subgroup $P$, then so does $F$.
\end{finitefact}
